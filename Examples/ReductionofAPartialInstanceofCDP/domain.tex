% Define commonly used style for action
\tikzset{
    STYLE/.style n args={3}{
        action={#1}{#2}{#3},
        minimum width=1cm,
        minimum height=1cm,
        rounded corners=0pt,
        fill={rgb,255:red,255; green,255; blue,255}
    }
}

\tikzset{
    % init and goal:
    INIT/.style = {
        init={},
        minimum height=8cm,
        fill={rgb,255:red,85; green,159; blue,184}
        },
    GOAL/.style = {
        goal={$l_1$,$l_2$,$l_3$,{...},$l_n$},
        minimum height=8cm,
        fill={rgb,255:red,85; green,159; blue,184}
        },
    % normal actions:
    A1/.style = {
        STYLE={}{$l_1$,$\neg l_2$ $\neg l_3$...}{a1}
        },
    A2/.style = {
        STYLE={}{$l_2$,$\neg l_3$...}{a2}
        },
    A3/.style = {
        STYLE={}{$l_3$}{a3}
        },
    AN/.style = {
        STYLE={}{$l_n$,...}{$a_n$}
        },
    PS/.style = {
        STYLE={}{$l_1...l_n$}{$ps^*$}
        },
    BMU/.style = {
        STYLE={$l_2$}{$l_n$}{$b_{\mu}$}
        },
    CMU/.style = {
        STYLE={$l_n$}{$l_2$}{$c_{\mu}$}
        }
}
