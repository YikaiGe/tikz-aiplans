% Causal link endpoint mark. Default is a filled circle.
% Place [causal mark=X] AFTER [causalLink] in the style list.
%   causal mark=*    filled 5-pointed star
%   causal mark=@    hollow circle
%   causal mark=^    filled triangle
%   causal mark={&}  filled square  (braces required around &)
%   causal mark=-    empty (no mark drawn)
%   default          filled circle (also restored by every causalLink)
\def\aiplansdotmark{\fill circle[radius=1pt];}
\tikzset{
  causal mark/.code={%
    \def\aiplansCMval{#1}%
    \def\aiplansCMstar{*}%
    \def\aiplansCMopen{@}%
    \def\aiplansCMtri{^}%
    \def\aiplansCMsq{&}%
    \def\aiplansCMnone{-}%
    \ifx\aiplansCMval\aiplansCMnone
      \gdef\aiplansdotmark{}%
    \else\ifx\aiplansCMval\aiplansCMstar
      \gdef\aiplansdotmark{%
        \fill (90:2pt)--(126:.8pt)--(162:2pt)--(198:.8pt)%
              --(234:2pt)--(270:.8pt)--(306:2pt)--(342:.8pt)%
              --(18:2pt)--(54:.8pt)--cycle;}%
    \else\ifx\aiplansCMval\aiplansCMopen
      \gdef\aiplansdotmark{\fill[white] circle[radius=2pt];\draw circle[radius=2pt];}%
    \else\ifx\aiplansCMval\aiplansCMtri
      \def\aiplansCMplv{vertical}%
      \ifx\aiplansplacement\aiplansCMplv
        \gdef\aiplansdotmark{\fill (0:2.5pt)--(120:2.5pt)--(240:2.5pt)--cycle;}%
      \else
        \gdef\aiplansdotmark{\fill (90:2.5pt)--(210:2.5pt)--(330:2.5pt)--cycle;}%
      \fi
    \else\ifx\aiplansCMval\aiplansCMsq
      \gdef\aiplansdotmark{\fill (-1.5pt,-1.5pt) rectangle (1.5pt,1.5pt);}%
    \else
      \gdef\aiplansdotmark{\fill circle[radius=1pt];}%
    \fi\fi\fi\fi\fi
  }%
}

% definition of causal links
\tikzset{
  causalLink/.style={
    causal mark=default,
    ->,
    shorten <= 0.15cm,
    shorten >= 0.15cm,
    postaction={
      decorate,
      decoration={
        markings,
        % Not using exaxtly 0 and 1 is a hack sugested by chatGPT
        % because otherwise for long links the goal dots are not placed
        mark=at position 0.0001 with {\pgftransformresetnontranslations\aiplansdotmark},
        mark=at position 0.9999 with {\pgftransformresetnontranslations\aiplansdotmark}
      }
    }
  },
  causalLink/.default={},
  % Version with label argument
  causalLink with label/.style={
    causalLink,
    postaction={
      decorate,
      decoration={
        markings,
        mark=at position 0.5 with {\node[above, font=\scriptsize] {#1};}
      }
    }
  }
}

\def\aiplanssetupperline#1{%
  \def\aiplanstmpplA{#1}%
  \ifx\aiplanstmpplA\empty
    \gdef\aiplansperlinect{0}%
  \else
    \setsepchar{,}%
    \readlist\aiplanslenlist{#1}%
    \xdef\aiplansperlinect{\aiplanslenlistlen}%
  \fi
}


\def\aiplansgetperlen#1#2{%
  \xdef\aiplanslinelen{#2}%
  \ifnum\aiplansperlinect>0
    \ifnum#1>\aiplansperlinect\else
      \edef\aiplansperlineval{\aiplanslenlist[#1]}%
      \def\aiplansperlinemty{}%
      \ifx\aiplansperlineval\aiplansperlinemty\else
        \xdef\aiplanslinelen{\aiplansperlineval}%
      \fi
    \fi
  \fi
}

\def\aiplanscomputelen#1#2{%
  \global\aiplanslblmax=0pt
  \foreach \aiplansLi in {1,...,#1} {%
    \setbox0=\hbox{\aiplansmeasurefont\labellist[\aiplansLi]}%
    \global\aiplanslbltmp=\wd0\relax
    \ifdim\aiplanslbltmp>\aiplanslblmax
      \global\aiplanslblmax=\aiplanslbltmp\relax
    \fi
  }%
  \xdef\aiplansautolblraw{\the\aiplanslblmax}%
  \global\aiplanslbltmp=#2\relax
  \ifdim\aiplanslbltmp>0pt
    \xdef\aiplansautolen{\the\aiplanslbltmp}%
  \else
    \global\advance\aiplanslblmax by 0.3cm\relax
    \xdef\aiplansautolen{\the\aiplanslblmax}%
  \fi
}

\def\aiplanscomputecoord#1#2{%
  \xdef#2{#1}%
}

% action={prec_labels}{eff_labels}{name}
%  Block size: minimum width=<w> (default 3cm), minimum height=<h>.
%  Label counts are auto-detected from the comma-separated lists.
%  Line lengths auto-sized; override with [prec label length=<dim>] / [eff label length=<dim>].
%  Placement: [horizontal] (default) precs left, effs right; [vertical] precs top, effs bottom.
%  Label side: [label side] places labels at line tips (precs left, effs right).

\tikzset{
    action/.style n args={3}{
        aiplans placement=horizontal,
        aiplans reset label side,
        draw,
        rounded corners,
        minimum width=3cm,
        fill=cyan,
        label={center:#3},
        append after command={
            \pgfextra{
                \edef\nodename{\tikzlastnode}
                \setsepchar{,}
                \def\aiplansplacementvert{vertical}
                \ifx\aiplansplacement\aiplansplacementvert
                % Vertical mode
                    \def\aiplanstmparg{#1}\ifx\aiplanstmparg\empty\else
                    \readlist\labellist{#1}
                    \aiplanscomputelen{\labellistlen}{\aiplansprelblovr}
                    \edef\aiplansprelen{\aiplansautolen}
                    \expandafter\aiplanssetupperline\expandafter{\aiplansprelblperline}%
                    \foreach \i in {1,...,\labellistlen} {
                        \aiplansgetperlen{\i}{\aiplansprelen}%
                        \coordinate (\nodename-pre-\i) at ([yshift=\aiplanslinelen] $(\nodename.north west)!\i/(\labellistlen+1)!(\nodename.north east)$);
                        \draw (\nodename.north -| \nodename-pre-\i) -- +(0,\aiplanslinelen) node [font=\scriptsize, aiplans vup opts] {\labellist[\i]};
                    }
                    \fi
                    \def\aiplanstmparg{#2}\ifx\aiplanstmparg\empty\else
                    \readlist\labellist{#2}
                    \aiplanscomputelen{\labellistlen}{\aiplansefflblovr}
                    \edef\aiplansefflen{\aiplansautolen}
                    \expandafter\aiplanssetupperline\expandafter{\aiplansefflblperline}%
                    \foreach \j in {1,...,\labellistlen} {
                        \aiplansgetperlen{\j}{\aiplansefflen}%
                        \coordinate (\nodename-eff-\j) at ([yshift=-\aiplanslinelen] $(\nodename.south west)!\j/(\labellistlen+1)!(\nodename.south east)$);
                        \draw (\nodename.south -| \nodename-eff-\j) -- +(0,-\aiplanslinelen) node [font=\scriptsize, aiplans vdown opts] {\labellist[\j]};
                    }
                    \fi
                \else
                % Horizontal mode (default)
                    \def\aiplanstmparg{#1}\ifx\aiplanstmparg\empty\else
                    \readlist\labellist{#1}
                    \aiplanscomputelen{\labellistlen}{\aiplansprelblovr}
                    \edef\aiplansprelen{\aiplansautolen}
                    \aiplanscomputecoord\aiplansprelen\aiplansprecoord
                    \expandafter\aiplanssetupperline\expandafter{\aiplansprelblperline}%
                    \foreach \i in {1,...,\labellistlen} {
                        \aiplansgetperlen{\i}{\aiplansprelen}%
                        \coordinate (\nodename-pre-\i) at ([xshift=-\aiplanslinelen] $(\nodename.north west)!\i/(\labellistlen+1)!(\nodename.south west)$);
                        \draw (\nodename.west |- \nodename-pre-\i) -- +(-\aiplanslinelen,0) node [font=\scriptsize, aiplans hleft opts] {\labellist[\i]};
                    }
                    \fi
                    \def\aiplanstmparg{#2}\ifx\aiplanstmparg\empty\else
                    \readlist\labellist{#2}
                    \aiplanscomputelen{\labellistlen}{\aiplansefflblovr}
                    \edef\aiplansefflen{\aiplansautolen}
                    \aiplanscomputecoord\aiplansefflen\aiplansefcoord
                    \expandafter\aiplanssetupperline\expandafter{\aiplansefflblperline}%
                    \foreach \j in {1,...,\labellistlen} {
                        \aiplansgetperlen{\j}{\aiplansefflen}%
                        \coordinate (\nodename-eff-\j) at ([xshift=\aiplanslinelen] $(\nodename.north east)!\j/(\labellistlen+1)!(\nodename.south east)$);
                        \draw (\nodename.east |- \nodename-eff-\j) -- +(\aiplanslinelen,0) node [font=\scriptsize, aiplans hright opts] {\labellist[\j]};
                    }
                    \fi
                \fi
            }
        }
    }
}
